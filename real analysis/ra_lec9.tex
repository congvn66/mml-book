\section{Some useful definitions}
\begin{definition}[Limit Points - recall]
A point $p$ is a limit point of $X$ if all neighbors (open balls) of $p$ contain (at least) a point $q\neq p$. To prove a point is not a limit point, find a neighbor that does not contain any $q \neq p$.
\end{definition}  

\begin{definition}[Discrete Metric Space]
$d(x,y)=1$ if $x\neq y$ else $d(x,y)=0$. 
\end{definition}

\begin{example}
Consider the metric space $(\RR, \text{discrete metric})$. This metric space doesn't have any limit points (there is a neighbor that doesn't contain any $q$ in the space that is different from our chosen point). Note that, an open ball in this space is 1 point, or all points in the space.
\end{example}

\begin{definition}[Isolated point]
A point $p$ is an isolated point of $E$ if $p \in E$ and $p$ is not a limit point of $E$.
\end{definition}

\begin{definition}[Interior point]
A point $p$ is an interior point of $E$ if there is a neighbor $N$ of $p$ such that $N \subset E$.

\textit{Or equivalently,} all neighbor $N$ of $p$ is not a subset of $E$.
\end{definition}

\begin{note}
With isolated point, imagine a point lies outside a circle (the circle is the set, the outside point is still in the set, but it is isolated). With interior point, imagine a point lives inside that non-border circle.
\end{note}

\section{Open and Closed Sets}

\begin{definition}[Open Sets]
A set $S$ is called open if all elements of $S$ are its interior points.
\end{definition}

\begin{example}
Here are some examples:
\begin{itemize}
    \item $(a,b)$ is an open set.
    \item $[c,d)$ is not an open set, because $c$ is not an interior point.
    \item $\emptyset$ is open.
    \item $\RR$ is open.
    \item Every subsets of $(\RR, \text{discrete metric})$ is open.
\end{itemize}
\end{example}

\begin{definition}[Closed sets]
$E$ is closed if it contains all its limit points.
\end{definition}

\begin{note}
Remember, limit points can be outside (not in) a set.

And open and closed will depend heavily on metric.

And open is not the opposite of closed.
\end{note}

\begin{example}
$\RR$ is closed, $[a,b]$ is closed, $[c,d)$ is not closed. And of course, the open ball is open, the closed ball is closed.
\end{example}

\begin{definition}[Closure]
$E'$ is the set of all limit points of $E$. The closure of $E$ is $\overline{E} = E' \cup E$. \textit{(spoiler: $\overline{E}$ is closed, closure of a open set make it closed)}.
\end{definition}

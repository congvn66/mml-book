\section{Some useful facts and relationship between Open and Closed sets}

\begin{theorem}
An open ball (neighborhood) is open.
\end{theorem}

\begin{theorem}
$\overline{E}$ (closure of $E$) is closed.
\end{theorem}

\begin{theorem}
$p$ is a limit point of $E$ $\Rightarrow$ every nbhd. of $p$ contains infinitely many points of $E$.
\end{theorem}

\begin{definition}[Dense]
$E$ is dense in $X$ if every points of $X$ is in $E$ or a limit point of $E$.

\textit{Or equivalently,} $\overline{E} = X$, or every ball in $X$ contains a point in $E$.
\end{definition}

\begin{example}
As you know, $\QQ$ is dense in $\RR$.
\end{example}

\begin{theorem}
$E$ closed $\iff$ $\overline{E} = E$.
\end{theorem}

\begin{theorem}
$E \subset \text{ closed set } F$, then $\overline{E} \subset F$.
\end{theorem}

\begin{theorem}
$E$ open $\iff$ $E^c$ closed.
\end{theorem}

\begin{note}
Those theorems are pretty easy to imagine.
\end{note}

\section{Union and Intersection of Open \& Closed sets}

\begin{lemma}
$\{E_{\alpha}\}$ is a collection of sets, then $(\bigcup_{\alpha}E_{\alpha})^c = \bigcap_{\alpha}E^c_{\alpha}$.
\end{lemma}

\begin{theorem}
Here are some theorems about union and intersection:
\begin{itemize}
    \item Arbitrary union of open sets is open.
    \item Arbitrary intersection of closed sets is closed.
    \item Finite union of closed sets is closed. (counter example for infinite case, union of $[-1+\frac{1}{n}, 1+\frac{1}{n}]$, the result will be $(-1,1)$ which is a open set)
    \item Finite intersection of open sets is open. (counter example intersection of $(\frac{-1}{n}, \frac{1}{n})$)
\end{itemize}
\end{theorem}

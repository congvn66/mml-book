\section{Convergence of sequences}
\begin{definition}[Sequences]
A sequence $\{p_n\}$ in a metric space $X$ is a function $f: \NN \rightarrow X$, mapping $n \rightarrow p_n \in X$.
\end{definition}

\begin{definition}[Range of a sequence]
The range of $\{p_n\}$ is the set $\{x: x=p_n \text{ for some } n\}$
\end{definition}

\begin{example}
$\operatorname{range}(\{1,2,3,4,5,4,4,5\}) = \{1,2,3,4,5\}$. You can interpret this as a value set of the given set.  
\end{example}

\begin{proposition}
A sequence is bounded if its range is bounded.
\end{proposition}

\begin{definition}
$\{p_n\}$ converges if $\exists p \in X$ s.t. $\forall \epsilon>0, \exists N\in\NN$ s.t. $n\ge N$ implies $d(p, p_n) < \epsilon$.
\end{definition}

\begin{note}
The definition makes us very hard to imagine what is happening right here. But it is pretty simple, a sequence that is converged if those "last" points are trapped inside a arbitrary chosen radius ball of $p$, as long as it started late enough ($n\ge N$). Or given a $\epsilon-$ball, you can always find a $N$ that every point after $p_N$ will be trapped inside that ball.

We denote $p_n$ converges to $p$ as: $p_n \rightarrow p$ or $\lim_{n \rightarrow \infty} p_n = p$.

And last, if you want to show $p_n \rightarrow p$, for each $\epsilon$, you must find a corresponding $N$ that \textbf{works}.
\end{note}

\section{Some propositions}
\begin{proposition}
$p_n \rightarrow p$ and $p_n \rightarrow p'$, then $p=p'$.
\end{proposition}
\begin{note}
The proof using contradiction, pretty straightforward.
\end{note}
\begin{note}
If a sequence is bounded, we can't imply it converges. You can imagine a sequence that jump from $a$ to $b$ back and forth. The sequence is of course bounded, but it doesn't converge.
\end{note}

\begin{proposition}
$p_n$ converges, then $p_n$ is bounded.
\end{proposition}

\begin{note}
By finite-ized the sequence by giving an $N$ correspond to  $\epsilon = 1$, we gonna use 
$$R = \max(1, d(p_1, p), d(p_2, p),..., d(p_{N-1},p))$$
then the sequence will be contain in the ball: $B_{R+1}(p)$, hence it is bounded. 
\end{note}

\begin{note}
If $p_n \rightarrow p$, we can't imply that, $p$ is the limit point of range of $\{p_n\}$. The counter example is a sequence with only 1 value.
\end{note}

\begin{proposition}
$p$ is a limit point of $E \subset X$, then there is a sequence $\{p_n\}$ in $E$ s.t. $p_n \rightarrow p$.
\end{proposition}

\begin{proposition}
$p_n \rightarrow p$ $\iff$ every nbhd. $U$ of $p$, $\{n\in\NN: p_n \not\in U\}$ is finite.
\end{proposition}

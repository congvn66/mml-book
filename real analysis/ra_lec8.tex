\section{Metric and Metric Space}

\begin{note}
I want to give an addition proposition for the last lecture about countable and uncountable sets, if $A \sim B$, we say that $A$ and $B$ have the same cardinality.
\end{note}

\begin{definition}[Metric Space]
A set $X$ is a metric space if there is a metric $d: X\times X \rightarrow \RR$ such that, for all $x, y \in X$:
\begin{itemize}
    \item \textbf{Non-negativity} $d(x,y) \ge 0$ and $d(x,y) = 0$ if and only if $x=y$.
    \item \textbf{Symmetric} $d(p,q) = d(q,p)$.
    \item \textbf{Triangle inequality} $d(p,q) \le d(p,r)+d(r,q) \forall r \in X$.
\end{itemize}
We say that $(X, d)$ is a metric space.
\end{definition}

\begin{example}
Just look at linear algebra or optimization.
\end{example}

\section{Balls and Limit points}

\begin{definition}[Open balls]
An open ball (neighbor) is a set: $N_r(x)=B(x,r)=\{y: d(x,y) < r\}$.
\end{definition}

\begin{definition}[Closed balls]
A closed ball is a set: $\overline{B(x,r)}=\{y: d(x,y) \le r\}$.
\end{definition}

\begin{definition}[Limit Points]
A point $p\in X$ is a limit point of $E$ if all open balls of $p$ contain a point $q\in X$ with $q\neq p$.
\end{definition}

\begin{note}
Note that, limit points are not border points.
\end{note}

\begin{note}
Last note, the metric in a space determines the shape of the open ball (Euclidean will be a round ball, but some metrics will be "rectangle" ball).
\end{note}

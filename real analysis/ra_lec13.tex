\section{Heine-Borel Theorem}
\begin{theorem}
$[a,b]$ is compact.
\end{theorem}

\begin{theorem}[Heine-Borel]
\textbf{In $\RR$} (or $\RR^n$), if $K$ is compact $\iff$ $K$ is closed and bounded.
\end{theorem}

\begin{theorem}
$K$ compact $\iff$ every infinite subset $E$ of $K$ has a limit point in $K$.
\end{theorem}

\begin{corollary}[Bolzano-Weierstrass]
Every bounded infinite subset of $\RR^k$ has a limit point in $\RR^k$.
\end{corollary}

\begin{theorem}[Generalization of intersection of nested intervals]
$K$ compact $\iff$ any collection of closed subsets $K_{\alpha}$ will have non-empty intersection.
\end{theorem}
